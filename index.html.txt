        function normalizeClientRecord(c) {
            if (!c.payment) {
                c.payment = {
                    status: 'paid',
                    fee: 6000,
                    mode: 'UPI / GPay',
                    date: c.startDate || new Date().toISOString().split('T')[0]
                };
            } else {
                c.payment.status = (c.payment.status === 'paid') ? 'paid' : 'unpaid';
                c.payment.fee = Number(c.payment.fee) || 0;
                c.payment.mode = c.payment.mode || 'UPI / GPay';
                c.payment.date = c.payment.date || c.startDate || '';
            }
            if (!Array.isArray(c.sessions) || c.sessions.length !== 12) {
                c.sessions = Array.from({ length: 12 }, (_, idx) => (c.sessions && c.sessions[idx]) || {
                    number: idx + 1,
                    completed: false,
                    date: idx === 0 ? (c.startDate || '') : '',
                    inTime: '',
                    outTime: '',
                    workout: ''
                });
            }
            return c;
        }

        // Initialize App with Firebase Cloud Sync + Local Fallback
        async function initApp() {
            applyModeLayout();
            loadLocalClients();

            // Connect to Firebase Firestore if config is available
            if (typeof __firebase_config !== 'undefined' && __firebase_config) {
                try {
                    const firebaseConfig = JSON.parse(__firebase_config);
                    const app = initializeApp(firebaseConfig);
                    auth = getAuth(app);
                    db = getFirestore(app);

                    if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                        await signInWithCustomToken(auth, __initial_auth_token);
                    } else {
                        await signInAnonymously(auth);
                    }
                    currentUser = auth.currentUser;
                    isCloudConnected = Boolean(currentUser && db);

                    if (isCloudConnected) {
                        startCloudSync();
                        return;
                    }
                } catch (err) {
                    console.error("Cloud sync init error, falling back to local mode:", err);
                }
            }

            // Fallback if offline or no cloud config
            updateConnectionBadge('local', 'Local Mode');
            if (!isClientView) {
                ensureSampleClientIfEmpty();
                activeClientId = clients[0]?.id || null;
                renderClientList();
            }
            renderMainWorkspace();
        }

        function loadLocalClients() {
            const savedData = localStorage.getItem(STORAGE_KEY);
            if (savedData) {
                try {
                    clients = JSON.parse(savedData).map(normalizeClientRecord);
                } catch (e) {
                    console.error("Failed to parse saved data", e);
                    clients = [];
                }
            }
        }

        function ensureSampleClientIfEmpty() {
            if (clients.length === 0) {
                const today = new Date().toISOString().split('T')[0];
                const sampleClient = createClientObject(

                clients.push(sampleClient);
                saveAllClientsToStorageAndCloud();
            }
        }

        function startCloudSync() {
            if (!db || !currentUser) return;
            if (unsubscribeCloud) unsubscribeCloud();

            if (isClientView && sharedClientId) {
                // Client Live Portal: Subscribe ONLY to this specific client's document
                const clientDocRef = doc(db, 'artifacts', appId, 'public', 'data', COLLECTION_NAME, sharedClientId);
                unsubscribeCloud = onSnapshot(
                    clientDocRef,
                    (docSnap) => {
                        updateConnectionBadge('live', 'Live Connected');
                        if (docSnap.exists()) {
                            singleClientData = normalizeClientRecord({ id: docSnap.id, ...docSnap.data() });
                        } else {
                            // Check local cache if opening on same device before sync
                            const localMatch = clients.find(c => c.id === sharedClientId);
                            singleClientData = localMatch ? normalizeClientRecord(localMatch) : null;
                        }
                        renderMainWorkspace();
                    },
                    (error) => {
                        console.error("Client live stream error:", error);
                        updateConnectionBadge('local', 'Offline Cache');
                        const localMatch = clients.find(c => c.id === sharedClientId);
                        singleClientData = localMatch || null;
                        renderMainWorkspace();
                    }
                );
            } else {
                // Trainer Dashboard: Subscribe to all clients in public collection
                const clientsColRef = collection(db, 'artifacts', appId, 'public', 'data', COLLECTION_NAME);
                let firstSnapshotHandled = false;

                unsubscribeCloud = onSnapshot(
                    clientsColRef,
                    async (snapshot) => {
                        updateConnectionBadge('live', 'Cloud Synced');
                        const cloudClients = [];
                        snapshot.forEach((docSnap) => {
                            cloudClients.push(normalizeClientRecord({ id: docSnap.id, ...docSnap.data() }));
                        });

                        // Sort newest first by createdAt or startDate
                        cloudClients.sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0));

                        if (!firstSnapshotHandled) {
                            firstSnapshotHandled = true;
                            if (cloudClients.length === 0 && clients.length > 0) {
                                // Migrate existing local clients to cloud automatically
                                for (const c of clients) {
                                    await syncClientToCloud(c);
                                }
                                return;
                            } else if (cloudClients.length === 0 && clients.length === 0) {
                                ensureSampleClientIfEmpty();
                                return;
                            }
                        }

                        clients = cloudClients;
                        localStorage.setItem(STORAGE_KEY, JSON.stringify(clients));

                        if (!activeClientId || !clients.some(c => c.id === activeClientId)) {
                            activeClientId = clients[0]?.id || null;
                        }

                        renderClientList();
                        // Avoid clobbering active textarea focus while trainer is typing
                        const activeTag = document.activeElement?.tagName;
                        if (activeTag !== 'TEXTAREA' && activeTag !== 'INPUT') {
                            renderMainWorkspace();
                        }
                    },
                    (error) => {
                        console.error("Trainer cloud sync error:", error);
                        updateConnectionBadge('local', 'Local Mode');
                        ensureSampleClientIfEmpty();
                        activeClientId = clients[0]?.id || null;
                        renderClientList();
                        renderMainWorkspace();
                    }
                );
            }
        }

        async function syncClientToCloud(client) {
            if (!db || !currentUser || !client || !client.id) return;
            try {
                updateConnectionBadge('saving', 'Syncing...');
                const clientDocRef = doc(db, 'artifacts', appId, 'public', 'data', COLLECTION_NAME, client.id);
                await setDoc(clientDocRef, JSON.parse(JSON.stringify(client)));
                updateConnectionBadge('live', 'Cloud Synced');
            } catch (err) {
                console.error("Failed to sync client to cloud:", err);
                updateConnectionBadge('local', 'Saved Locally');
            }
        }

        async function deleteClientFromCloud(clientId) {
            if (!db || !currentUser || !clientId) return;
            try {
                const clientDocRef = doc(db, 'artifacts', appId, 'public', 'data', COLLECTION_NAME, clientId);
                await deleteDoc(clientDocRef);
            } catch (err) {
                console.error("Failed to delete client from cloud:", err);
            }
        }

        // Helper to generate fresh 12-session client structure
        function createClientObject(name, phone, startDate, goal, payment = null) {
            const sessions = Array.from({ length: 12 }, (_, idx) => ({
                number: idx + 1,
                completed: false,
                date: idx === 0 ? startDate : '',
                inTime: '',
                outTime: '',
                workout: ''
            }));

            return {
                id: 'client_' + Date.now() + '_' + Math.random().toString(36).substr(2, 5),
                createdAt: Date.now(),
                name: name.trim(),
                phone: phone.replace(/\D/g, '').slice(-10),
                startDate: startDate,
                goal: goal,
                payment: payment || {
                    status: 'unpaid',
                    fee: 6000,
                    mode: 'UPI / GPay',
                    date: startDate
                },
                sessions: sessions
            };
        }

        // Save to LocalStorage & Cloud Firestore
        function saveToStorage(specificClient = null) {
            localStorage.setItem(STORAGE_KEY, JSON.stringify(clients));
            const targetClient = specificClient || clients.find(c => c.id === activeClientId);
            if (targetClient && isCloudConnected) {
                syncClientToCloud(targetClient);
            } else if (!isCloudConnected) {
                updateConnectionBadge('local', 'Saved Locally');
            }
        }

        async function saveAllClientsToStorageAndCloud() {
            localStorage.setItem(STORAGE_KEY, JSON.stringify(clients));
            if (isCloudConnected) {
                for (const c of clients) {
                    await syncClientToCloud(c);
                }
            }
        }

        // SESSION FIELD UPDATES (TRAINER ONLY)
        // ==========================================
        function toggleSessionComplete(sessionIdx, isChecked) {
            if (isClientView) return;
            const client = clients.find(c => c.id === activeClientId);
            if (!client) return;

            client.sessions[sessionIdx].completed = isChecked;
            if (isChecked && !client.sessions[sessionIdx].date) {
                client.sessions[sessionIdx].date = new Date().toISOString().split('T')[0];
            }

            saveToStorage(client);
            renderClientList();
            renderMainWorkspace();

            const completedCount = client.sessions.filter(s => s.completed).length;
            if (completedCount === 12 && isChecked) {
                showToast(`🏆 All 12 sessions completed for ${client.name}!`, 'success');
            } else {
                showToast(`Session #${sessionIdx + 1} marked ${isChecked ? 'Done ✓' : 'Pending'}`);
            }
        }

        function updateSessionField(sessionIdx, field, value) {
            if (isClientView) return;
            const client = clients.find(c => c.id === activeClientId);
            if (!client) return;

            client.sessions[sessionIdx][field] = value;
            localStorage.setItem(STORAGE_KEY, JSON.stringify(clients));

            // Debounce cloud writes while typing workout notes
            const timerKey = `${client.id}_${sessionIdx}_${field}`;
            clearTimeout(fieldDebounceTimers[timerKey]);
            updateConnectionBadge('saving', 'Syncing...');
            fieldDebounceTimers[timerKey] = setTimeout(() => {
                saveToStorage(client);
            }, 500);
        }

        function fillCurrentTime(sessionIdx, field) {
            if (isClientView) return;
            const now = new Date();
            const hh = String(now.getHours()).padStart(2, '0');
            const mm = String(now.getMinutes()).padStart(2, '0');
            const currentTime = `${hh}:${mm}`;

            const client = clients.find(c => c.id === activeClientId);
            if (!client) return;

            client.sessions[sessionIdx][field] = currentTime;
            if (!client.sessions[sessionIdx].date) {
                client.sessions[sessionIdx].date = now.toISOString().split('T')[0];
            }

            saveToStorage(client);
            renderMainWorkspace();
            showToast(`Set ${field === 'inTime' ? 'In Time' : 'Out Time'} to ${formatTime12Hour(currentTime)}`);
        }

        // Toast Notification
        function showToast(message, type = 'success') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            const bgColor = type === 'success' ? 'bg-emerald-600' : type === 'whatsapp' ? 'bg-gym-whatsapp text-black font-bold' : type === 'error' ? 'bg-red-600' : 'bg-gym-orange';
            
            toast.className = `${bgColor} text-white px-4 py-3 rounded-xl shadow-xl text-sm flex items-center gap-2.5 transform transition-all duration-300 translate-y-2 opacity-0`;
            toast.innerHTML = `<i class="fa-solid ${type === 'whatsapp' ? 'fa-brands fa-whatsapp text-lg' : type === 'error' ? 'fa-circle-exclamation' : 'fa-circle-check'}"></i> <span>${message}</span>`;
            
            container.appendChild(toast);
            requestAnimationFrame(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            });

            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-2');
                setTimeout(() => toast.remove(), 300);
            }, 3200);
        }

        // ==========================================
        // SHARE CLIENT LIVE PORTAL LINK VIA WHATSAPP

        function getClientLiveUrl(clientId) {
            const url = new URL(window.location.href);
            url.search = '';
            url.hash = '';
            url.searchParams.set('client', clientId);
            return url.toString();
        }

        function copyTextToClipboard(text) {
            const tempInput = document.createElement('textarea');
            tempInput.value = text;
            tempInput.style.position = 'fixed';
            tempInput.style.opacity = '0';
            document.body.appendChild(tempInput);
            tempInput.select();
            try {
                document.execCommand('copy');
            } catch (e) {
                console.error('Copy failed', e);
            }
            document.body.removeChild(tempInput);
        }

        // ==========================================
        // PAYMENT STATUS TOGGLE & WHATSAPP RECEIPT/REMINDER
        // ==========================================
        function toggleClientPaymentStatus() {
            if (isClientView) return;
            const client = clients.find(c => c.id === activeClientId);
            if (!client) return;

            if (!client.payment) {
                client.payment = { status: 'unpaid', fee: 6000, mode: 'UPI / GPay', date: '' };
            }

            const becomingPaid = client.payment.status !== 'paid';
            client.payment.status = becomingPaid ? 'paid' : 'unpaid';
            if (becomingPaid && !client.payment.date) {
                client.payment.date = new Date().toISOString().split('T')[0];
            }

            saveToStorage(client);
            renderClientList();
            renderMainWorkspace();

            if (becomingPaid) {
                showToast(`✅ Payment received! ${client.name}'s PT plan is now ACTIVE.`);
            } else {
                showToast(`⚠️ Status changed to UNPAID (Inactive) for ${client.name}.`);
            }
        }

            const cleanPhone = phone.replace(/\D/g, '');
            const fullPhone = cleanPhone.length === 10 ? `91${cleanPhone}` : cleanPhone;

            window.open(url, '_blank');
        }

            e.preventDefault();

            const name = document.getElementById('client-name').value;
            const phone = document.getElementById('client-phone').value;

            const paymentStatus = document.getElementById('status-paid').checked ? 'paid' : 'unpaid';
            const paymentFee = Number(document.getElementById('client-fee').value) || 0;

            const paymentDate = document.getElementById('client-payment-date').value;

            const paymentObj = {
                status: paymentStatus,
                fee: paymentFee,
                mode: paymentMode,
                date: paymentDate
            };

            let savedClient = null;
            if (editId) {
                const client = clients.find(c => c.id === editId);
                if (client) {
                    client.name = name.trim();
                    client.phone = phone.replace(/\D/g, '').slice(-10);
                    client.startDate = startDate;

                    client.payment = paymentObj;
                    savedClient = client;
                    showToast(`Updated ${client.name} (${paymentStatus === 'paid' ? 'ACTIVE • PAID' : 'UNPAID'})`);
                }
            } else {
                const newClient = createClientObject(name, phone, startDate, goal, paymentObj);
                clients.unshift(newClient);
                activeClientId = newClient.id;
                savedClient = newClient;
                showToast(`Added ${newClient.name} (${paymentStatus === 'paid' ? 'ACTIVE • PAID' : 'UNPAID'})`);
            }

            saveToStorage(savedClient);
            closeClientModal();
            renderClientList();
            renderMainWorkspace();
        }

        function exportData() {
            const dataStr = JSON.stringify(clients, null, 2);
            const blob = new Blob([dataStr], { type: 'application/json' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');

            a.href = url;

            showToast('Backup file downloaded!');
        }

        function importData(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = async function(e) {
                try {
                    const imported = JSON.parse(e.target.result);
                    if (Array.isArray(imported)) {
                        clients = imported.map(normalizeClientRecord);
                        activeClientId = clients.length > 0 ? clients[0].id : null;
                        await saveAllClientsToStorageAndCloud();
                        renderClientList();
                        renderMainWorkspace();
                        showToast(`Restored & synced ${clients.length} client records!`);
                    } else {
                        showToast('Invalid backup file format', 'error');
                    }
                } catch (err) {
                    showToast('Error reading backup file', 'error');
                }
            };
            reader.readAsText(file);
            event.target.value = '';
        }

        // ==========================================
        // FORMATTING HELPERS
        // ==========================================
        function formatDisplayDate(isoDate) {
            if (!isoDate) return '';
            const parts = isoDate.split('-');
            if (parts.length !== 3) return isoDate;
            const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
            return `${parts[2]} ${months[parseInt(parts[1], 10) - 1]} ${parts[0]}`;
        }

        function formatTime12Hour(time24) {
            if (!time24) return '';
            const [hStr, mStr] = time24.split(':');
            let h = parseInt(hStr, 10);
            const ampm = h >= 12 ? 'PM' : 'AM';
            h = h % 12 || 12;
            return `${h}:${mStr} ${ampm}`;
        }

        function escapeHtml(text) {
            if (!text) return '';
            return text
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#039;");
        }
